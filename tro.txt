  document.getElementById('loginUser').value = u;
  document.getElementById('loginPass').value = p;
}

  e.preventDefault();
  const u = document.getElementById('loginUser').value.trim();
  const p = document.getElementById('loginPass').value.trim();
  const user = DB.NGUOI_DUNG.find(x => x.TenDangNhap === u && x.MatKhau === p);
  const errBox = document.getElementById('loginError');

  if (!user) {
    errBox.textContent = 'Sai tên đăng nhập hoặc mật khẩu trong bảng NGUOI_DUNG!';
    errBox.classList.remove('hidden');
    return;
  }
  errBox.classList.add('hidden');
  currentUser = user;
  sessionStorage.setItem('QL_USER', JSON.stringify(user));
  initApp();
});

function logout() {
  currentUser = null;
  sessionStorage.removeItem('QL_USER');
  document.getElementById('appScreen').classList.add('hidden');
  document.getElementById('loginScreen').classList.remove('hidden');
}

function initApp() {

  document.getElementById('loginScreen').classList.add('hidden');
  document.getElementById('appScreen').classList.remove('hidden');
  document.getElementById('currentUserName').textContent = currentUser.HoTen;
  document.getElementById('currentUserRole').textContent = `${currentUser.VaiTro} (${currentUser.TenDangNhap})`;
  renderSidebar();
  renderMain();
}

function formatMoney(n) {
  return Number(n || 0).toLocaleString('vi-VN') + ' đ';
}

function resolveFKDisplay(field, val) {
  if (field.type !== 'fk') return val;
  const refRow = DB[field.refTable].find(r => r[field.refKey] === val);
  if (!refRow) return `<span class="text-red-500">${val} (Lỗi FK)</span>`; const extra = field.refTable === 'PHONG' ? `Phòng ${refRow.SoPhong}` : refRow[field.refLabel];
  return `<span class="inline-flex items-center gap-1 px-2 py-0.5 rounded bg-blue-50 text-blue-700 border border-blue-200 text-xs font-medium">
    <b>${val}</b> ·${extra}
  </span>`;
}
